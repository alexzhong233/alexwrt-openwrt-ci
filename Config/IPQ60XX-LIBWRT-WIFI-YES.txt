# 京东云雅典娜 RE-CS-02 —— LibWrt 25.12-nss 试编版
#
# 与 IPQ60XX-OWRT-WIFI-YES.txt 的唯一差别是 NSS 相关项：两个上游的
# NSS 实现完全不同，包名和 Kconfig 符号都不一样，不能混用。
#
#   VIKINGYFY/immortalwrt@owrt   nss-wifi-drv（自研驱动 + wifili 用户态）
#                                CONFIG_ATH11K_NSS_WIFI
#                                debugfs: /sys/kernel/debug/39000000.nss/
#
#   LiBwrt/LibWrt@25.12-nss      qosmio 那套 qca-nss-drv + qca-nss-ecm
#                                （feed: nss_packages，分支 NSS-LibWrt）
#                                CONFIG_ATH11K_NSS_SUPPORT
#                                ath.mk 里 MODPARAMS.ath11k:=nss_offload=1 frame_mode=2
#                                debugfs: /sys/kernel/debug/qca-nss-drv/、/ecm/
#
# 两边都是真正的 WiFi 卸载（LibWrt 那套靠 package/kernel/mac80211/patches/nss
# 的 ath11k 补丁 + nss-drv 的 wifili 转发），不是「只卸有线」。
#
# 注意：本文件只用于试编对比。luci-app-nss-status 读的是 nss-wifi-drv 的
# debugfs 布局，在 LibWrt 上会显示「未就绪」—— 那是预期行为，不代表 NSS 没工作，
# 要确认得看 /sys/kernel/debug/qca-nss-drv/。

# NSS：qca-nss-drv + qca-nss-ecm。
#   · ATH11K_NSS_SUPPORT 打开后，ath.mk 会自动带上 kmod-qca-nss-drv 依赖、
#     把 ath11k 的模块参数设成 nss_offload=1 frame_mode=2，并 select
#     PACKAGE_kmod-qca-nss-ecm —— 所以理论上只写这一行就够，下面几个显式写出
#     来是为了让「到底选中了什么」在 .config 里一眼可见，也防止上游改了
#     Kconfig 结构后悄悄少装东西。
#   · nss-firmware 是 meta 包，按 target 自动依赖 nss-firmware-ipq60xx。
#     固件版本跟随上游默认（11_4，支持 802.11s mesh）；要 12.5 就加
#     CONFIG_NSS_FIRMWARE_VERSION_12_5=y。
CONFIG_ATH11K_NSS_SUPPORT=y
CONFIG_PACKAGE_nss-firmware=y
CONFIG_PACKAGE_kmod-qca-nss-drv=y
CONFIG_PACKAGE_kmod-qca-nss-ecm=y
# 状态页照装：包本身编得出来，只是读不到 nss-wifi-drv 的 debugfs。
# 留着正好用来确认「换了上游之后页面确实拿不到数」，别让它默默消失。
CONFIG_PACKAGE_luci-app-nss-status=y
# 这边真正的状态页：yduke/luci-app-nss 读 /sys/kernel/debug/qca-nss-drv/stats
# 和 /sys/kernel/debug/ecm/，显示 NSS 核心负载、各模块收发计数、ECM 加速流数，
# 并在「状态 → 概览」插一个 NSS 区块。JS 版 LuCI，只依赖 luci-base。
# 配套 luci-app-nssecm 是 ECM 的配置表单（acceleration_engine 可选
# auto / nss / sfe / ppe / nss-sfe），改完重启 /etc/init.d/qca-nss-ecm。
CONFIG_PACKAGE_luci-app-nss=y
CONFIG_PACKAGE_luci-app-nssecm=y

# ── 数据面扩容：只补「真能吃到卸载」的三块 ────────────────────────────
# 判断依据是 qca-nss-drv/Config.in：所有 NSS_DRV_*_ENABLE 默认都是 n，只有
# 被 select 或显式写 y 才编进 drv。而 clients 里每个 .ko 的 DEPENDS 都写着
# +@NSS_DRV_XXX_ENABLE —— 两层开关，少一层模块直接不可见，写了也会被丢掉。
#
# 桥接（BRIDGE）：LAN 内互访（WiFi↔有线、有线↔有线、NAS↔PC）是 L2 转发，
#   既不经过代理也不出 WAN，是这台机器上唯一「开了就真省 CPU」的流量。
CONFIG_NSS_DRV_BRIDGE_ENABLE=y
CONFIG_PACKAGE_kmod-qca-nss-drv-bridge-mgr=y
# VLAN：802.1q 子接口。bridge-mgr 的 DEPENDS 会自动带 vlan-mgr，显式写出来
#   是为了 .config 里一眼能核对。
CONFIG_NSS_DRV_VLAN_ENABLE=y
CONFIG_PACKAGE_kmod-qca-nss-drv-vlan-mgr=y
# PPPoE：家宽拨号时 WAN 侧所有流量都挂在 ppp 接口上。drv 侧的
#   NSS_DRV_PPPOE_ENABLE 只有 kmod-pppoe 被选中时才会被 ecm 的 DEPENDS 带起来，
#   而 clients 的 qca-nss-pppoe.ko 没有任何包会自动选 —— 缺它 ECM 就没有
#   PPPoE accel engine，只能加速 LAN 内的 L3 流，等于白开。三行都写死。
CONFIG_PACKAGE_kmod-pppoe=y
CONFIG_NSS_DRV_PPPOE_ENABLE=y
CONFIG_PACKAGE_kmod-qca-nss-drv-pppoe=y
#
# 以下刻意不补，开了也是白占内存：
#   · IPSEC / TLS / DTLS / QVPN —— 只对「路由器自己终结」的连接有效。
#     客户端↔外网的 HTTPS 是端到端加密，NSS 看不到明文；VPN 同理。
#     真正跑在隧道里的是 netbird(WireGuard)，NSS 根本没有 WireGuard accel，
#     而 IPSEC 那套是给 strongSwan 用的，和 wt0 无关。
#   · L2TP / PPTP / GRE / VXLAN / MAP-T / TUN6RD / TUNIPIP6 —— 没有任何一条在用。
#   · SHAPER / IGS —— 没做 SQM 限速，也不看 IPTV 组播。
#   · LAG —— 没做链路聚合。
#   · WIFI_MESH —— 走的是 AP 不是 802.11s。
#
# docker / uu / nikki / passwall 这四类本来就不可能卸载，别指望：
#   docker 的 veth 是纯软件设备，NSS 不认识；uu 和代理靠 tproxy + mark 打标，
#   ECM 见到标记就主动跳过。它们只会占用 conntrack，不会占用 NSS 连接表。
