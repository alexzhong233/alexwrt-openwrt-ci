# 雅典娜个人固件的最终覆盖项。
# WRT-CORE 在设备配置和 GENERAL.txt 之后追加本文件，因此这里的值优先。

# 编译 HomeProxy 和 PassWall2，但不默认安装到固件 rootfs；需要时可从软件包管理器安装。
CONFIG_PACKAGE_luci-app-homeproxy=m
CONFIG_PACKAGE_luci-app-passwall2=m

# 与当前路由器软件清单保持一致：使用 diskman，不额外安装 mini-diskmanager。
CONFIG_PACKAGE_luci-app-diskman=y
CONFIG_PACKAGE_luci-app-mini-diskmanager=n

# NSS 走上游的 wifi 侧驱动（wifili 直通无线帧）。
# CONFIG_ATH11K_NSS_WIFI 决定 /usr/share/nss-wifi/wlan 是否为 1，缺了它启动脚本
# 会认为没有可卸载的 WiFi 而退回 host path；固件线选 12.5，跑 802.11s mesh 才换 11.4。
CONFIG_PACKAGE_nss-firmware=y
CONFIG_PACKAGE_kmod-nss-wifi-drv=y
CONFIG_ATH11K_NSS_WIFI=y
CONFIG_NSS_FIRMWARE_VERSION_12_5=y
